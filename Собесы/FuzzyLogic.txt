        #!/usr/bin/env python3
        import sys
        blockSize = 10
        delimiters = 9*[','] + ['\n']
        while block := sys.stdin.read(blockSize):
          delimiters.append(ifs := delimiters.pop(0))
          sys.stdout.write(block + ifs)
